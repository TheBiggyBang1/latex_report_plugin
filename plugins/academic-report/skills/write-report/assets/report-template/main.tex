% \crispdm{ID}: marks the section that addresses CRISP-DM generic task ID

% \crispdm[na]{ID}: the task does not apply; the section explains why.

    \expandafter\gdef\csname crispdm@seen@#2\endcsname{}\label{crisp:#2}%
  \fi}
